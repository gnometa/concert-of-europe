#########################################################
#                                                       #
#   The Peru-Bolivian Confederation, 1835-1839 (GVG)    #
#   ids 1002900-1002907                                 #
#                                                       #
#   Layers the missing 1835-1839 history onto the PDM   #
#   apparatus that already exists rather than inventing #
#   a tag or a parallel formation path: the PBC/SPU/NPU #
#   tags, decisions/PBC.txt (form_PBC_BOL,              #
#   form_PBC_PEU, pbc_in_chaos, end_of_confederation_*) #
#   and events/PBCFlavor.txt 97040-97055.               #
#   This file supplies the march of Santa Cruz on Peru, #
#   the Chilean War of the Restoration (the war         #
#   declaration ChileanEvents.txt 97050 gave up on),    #
#   Rosas, Paucarpata and Yungay. It neither forms nor  #
#   dissolves the Confederation - the existing          #
#   decisions and 97055 do both.                        #
#                                                       #
#   Bolivia does not exist at the 1821 start and no     #
#   other file in the mod ever creates her: every Upper #
#   Peruvian province is held by the royalists (or by   #
#   Lima), and every release_vassal = BOL /             #
#   change_tag = BOL site sits inside the dissolution   #
#   of a Confederation that cannot form without her.    #
#   1002906 and 1002907 therefore give Bolivia her      #
#   scripted birth (Chuquisaca, 1825), fired for        #
#   whoever holds Chuquisaca rather than for a fixed    #
#   tag, so the chain works under either owner of Upper #
#   Peru.                                               #
#                                                       #
#   No fire_only_once anywhere: it is engine-wide, not  #
#   per country. Every self-firing event is guarded by  #
#   a country flag set in EVERY option.                 #
#                                                       #
#########################################################

# The Congress of Chuquisaca - the birth of Bolivia, 1825
# Fired for whoever holds Chuquisaca (the royalists at the 1821 start), never
# for a hard-coded tag: nothing else in the mod creates BOL, and everything
# below this line needs her.
country_event = {
	id = 1002906
	title = "EVTNAME1002906"
	desc = "EVTDESC1002906"
	picture = "national_congress"

	trigger = {
		owns = 2313 #Chuquisaca
		year = 1825
		NOT = { year = 1836 }
		# NOR: none of these four may hold
		NOT = {
			tag = BOL
			exists = BOL
			has_global_flag = bolivia_independent
			has_country_flag = chuquisaca_answered
		}
	}

	mean_time_to_happen = {
		months = 3
		modifier = {
			factor = 0.5
			NOT = { owns = 2295 } #Lima
		}
		modifier = {
			factor = 0.75
			war = yes
		}
	}

	# Recognise the Republic of Bolivar
	option = {
		name = "EVTOPTA1002906"
		set_country_flag = chuquisaca_answered
		set_global_flag = bolivia_independent
		prestige = -2
		# secede_province creates the tag, as vanilla Taiping.txt 160001 does
		any_owned = {
			limit = { is_core = BOL }
			secede_province = BOL
		}
		random_country = {
			limit = {
				tag = BOL
				exists = yes
			}
			relation = { who = THIS value = 100 }
		}
		ai_chance = {
			factor = 60
		}
	}

	# The sierra will be held
	option = {
		name = "EVTOPTB1002906"
		set_country_flag = chuquisaca_answered
		set_country_flag = upper_peru_held
		prestige = 2
		any_owned = {
			limit = { is_core = BOL }
			any_pop = {
				militancy = 3
				consciousness = 1
			}
			add_province_modifier = {
				name = nationalist_agitation
				duration = 1095
			}
		}
		ai_chance = {
			factor = 40
		}
	}
}

# Santa Cruz Answers Orbegoso
country_event = {
	id = 1002900
	title = "EVTNAME1002900"
	desc = "EVTDESC1002900"
	picture = "peru-bolivia"
	news = yes
	# Option-neutral: the article is published whichever option is taken,
	# so it reports the appeal and the debate, not a march option C refuses
	news_title = "EVTNEWS1002900_TITLE"
	news_desc_long = "EVTNEWS1002900_LONG"
	news_desc_medium = "EVTNEWS1002900_MEDIUM"
	news_desc_short = "EVTNEWS1002900_SHORT"

	trigger = {
		tag = BOL
		exists = yes
		is_vassal = no
		# Degrade gracefully: whichever half of the altiplano Bolivia was
		# born with in 1002906/1002907 is enough to answer Orbegoso
		OR = {
			owns = 2310 #La Paz
			owns = 2313 #Chuquisaca
		}
		war = no
		year = 1835
		NOT = { year = 1841 }
		PEU = {
			exists = yes
			owns = 2295 #Lima
			is_vassal = no
			war = no
		}
		# NOR: none of these five may hold
		NOT = {
			truce_with = PEU
			has_global_flag = peru_bolivia_confederated
			has_country_flag = wants_to_unify
			has_country_flag = santa_cruz_decided
			exists = PBC
		}
	}

	mean_time_to_happen = {
		months = 4
		modifier = {
			factor = 0.5
			PEU = { num_of_revolts = 1 }
		}
		modifier = {
			factor = 0.75
			PEU = {
				any_owned_province = { average_militancy = 4 }
			}
		}
	}

	# March on Peru: sets the flag form_PBC_BOL and PBCFlavor 97045 key off
	option = {
		name = "EVTOPTA1002900"
		set_country_flag = santa_cruz_decided
		set_country_flag = wants_to_unify
		prestige = 2
		add_country_modifier = {
			name = small_country_draft
			duration = 730
		}
		any_pop = {
			dominant_issue = {
				factor = 0.03
				value = jingoism
			}
		}
		war = {
			target = PEU
			attacker_goal = {
				casus_belli = cut_down_to_size
			}
			defender_goal = {
				casus_belli = humiliate
			}
		}
		PEU = { country_event = { id = 1002901 days = 10 } }
		ai_chance = {
			factor = 70
			modifier = {
				factor = 0
				war = yes
			}
			modifier = {
				factor = 0
				truce_with = PEU
			}
		}
	}

	# Armed mediation
	option = {
		name = "EVTOPTB1002900"
		set_country_flag = santa_cruz_decided
		prestige = 1
		relation = { who = PEU value = 50 }
		diplomatic_influence = { who = PEU value = 25 }
		add_country_modifier = {
			name = usa_draft
			duration = 365
		}
		ai_chance = {
			factor = 20
		}
	}

	# Bolivia stays out of Peru
	option = {
		name = "EVTOPTC1002900"
		set_country_flag = santa_cruz_decided
		prestige = -2
		any_pop = { militancy = -1 }
		random_country = {
			limit = {
				tag = CHL
				exists = yes
			}
			relation = { who = THIS value = 50 }
		}
		random_country = {
			limit = {
				tag = ARG
				exists = yes
			}
			relation = { who = THIS value = 50 }
		}
		ai_chance = {
			factor = 10
		}
	}
}

# The Battle of Yungay
country_event = {
	id = 1002905
	title = "EVTNAME1002905"
	desc = "EVTDESC1002905"
	picture = "war_ended"
	major = yes
	news = yes
	news_title = "EVTNAME1002905_NEWS_TITLE"
	news_desc_long = "EVTDESC1002905_NEWS_LONG"
	news_desc_medium = "EVTDESC1002905_NEWS_MEDIUM"
	news_desc_short = "EVTDESC1002905_NEWS_SHORT"

	trigger = {
		tag = CHL
		exists = yes
		war_with = PBC
		has_country_flag = restoration_war_declared
		# war_score keeps this reachable once the Peruvian vassals of
		# PBC have broken away with Lima and Huaraz (pbc_in_chaos)
		OR = {
			2298 = { controlled_by = THIS } #Huaraz
			2295 = { controlled_by = THIS } #Lima
			war_score = 25
		}
		NOT = { has_country_flag = yungay_won }
	}

	mean_time_to_happen = {
		months = 3
	}

	# Restore the three republics
	option = {
		name = "EVTOPTA1002905"
		set_country_flag = yungay_won
		prestige = 10
		badboy = 1
		# Deliberately does not dissolve PBC: PBCFlavor.txt 97055,
		# pbc_in_chaos and end_of_confederation_* already do that
		PBC = {
			prestige = -10
			war_exhaustion = 15
			add_country_modifier = {
				name = national_instability
				duration = 1095
			}
		}
		# On the form_PBC_BOL path the Confederation owns no Peruvian core at
		# all - PEU and SPU were released as vassals - so the agitation has to
		# run over the Confederation and its vassals together
		any_country = {
			limit = {
				OR = {
					tag = PBC
					vassal_of = PBC
				}
			}
			any_owned = {
				limit = { is_core = PEU }
				any_pop = {
					militancy = 4
					consciousness = 2
				}
				add_province_modifier = {
					name = nationalist_agitation
					duration = 1095
				}
			}
		}
		random_country = {
			limit = {
				tag = PEU
				exists = yes
			}
			relation = { who = THIS value = 100 }
		}
		random_country = {
			limit = {
				tag = NPU
				exists = yes
			}
			relation = { who = THIS value = 100 }
		}
		random_country = {
			limit = {
				tag = SPU
				exists = yes
			}
			relation = { who = THIS value = 100 }
		}
		random_country = {
			limit = {
				tag = BOL
				exists = yes
			}
			relation = { who = THIS value = 100 }
		}
		ai_chance = {
			factor = 70
		}
	}

	# Take the settlement and go home
	option = {
		name = "EVTOPTB1002905"
		set_country_flag = yungay_won
		prestige = 5
		end_war = PBC
		PBC = {
			prestige = -5
			war_exhaustion = 5
		}
		relation = { who = PBC value = 50 }
		ai_chance = {
			factor = 30
		}
	}
}
