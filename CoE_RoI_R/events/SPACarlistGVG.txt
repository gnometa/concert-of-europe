#########################################################
#                                                       #
#      The First Carlist War, 1830-1840 (GVG)           #
#      ids 1002700-1002705                              #
#                                                       #
#  Dates the Carlist succession crisis that             #
#  events/SPAFlavor.txt already scripts in full         #
#  (37760/37710/37711/37712/37713-37717 plus the        #
#  carlist_rebels and christino_rebels types). No       #
#  legacy file is edited: 1002700 pre-empts 37760 by    #
#  setting carlism_questioned itself and holding        #
#  national_instability as a lock, and 1002701 lifts    #
#  that lock on the day Ferdinand VII dies, which       #
#  releases legacy 37711 (MTTH 1 day) on 1833.9.30      #
#  with its SPA/SPC fork and Espartero intact.          #
#                                                       #
#  War state is tracked by this chain's OWN flag        #
#  gvg_carlist_war, set in every option of 1002701.     #
#  The legacy flag carlist_war cannot be relied on:     #
#  37711 is fire_only_once and only its option A sets   #
#  it, while 37712 clears it on a 1-month MTTH in any   #
#  revolt-free lull and nothing can set it again.       #
#  1002704 clears carlist_war in BOTH options so the    #
#  legacy 37712 ending, and its hidden                  #
#  remove_accepted_culture = basque, cannot fire on     #
#  top of ours.                                         #
#                                                       #
#  The legacy tally carlist_provinces is SET to 6 by    #
#  37711 one day after 1002701, so 1002701 does not     #
#  touch it; the three branches differ through the      #
#  carlist_sympathies province modifiers they place     #
#  on 494/496/504, which 37711 does not overwrite.      #
#                                                       #
#  month is 0-indexed. The month gates are written as   #
#  OR = { year = N+1 AND = { year = N month = M } }     #
#  because month = M means M or later in the year.      #
#                                                       #
#########################################################

# The Pragmatic Sanction
country_event = {
	id = 1002700
	title = "EVTNAME1002700"
	desc = "EVTDESC1002700"
	picture = "Monarchy"
	major = yes

	trigger = {
		tag = SPA
		exists = yes
		OR = {
			year = 1830
			AND = {
				year = 1829
				month = 11 #December
			}
		}
		NOT = { year = 1831 }
		OR = {
			OR = { government = hms_government government = hms_government2 government = hms_government3 }
			OR = { government = prussian_constitutionalism government = prussian_constitutionalism2 government = prussian_constitutionalism3 }
			OR = { government = absolute_monarchy government = absolute_monarchy2 government = absolute_monarchy3 }
		}
		NOT = { has_country_flag = carlism_questioned }
	}

	mean_time_to_happen = {
		days = 20
	}

	option = {
		name = "EVTOPTA1002700"
		set_country_flag = carlism_questioned
		set_country_flag = pragmatic_sanction_1830
		prestige = 2
		add_country_modifier = {
			name = national_instability
			duration = 1825
		}
		any_pop = {
			limit = {
				OR = {
					type = clergymen
					type = aristocrats
				}
			}
			militancy = 2
			consciousness = 1
			ideology = {
				value = reactionary
				factor = 0.05
			}
		}
		any_owned = {
			limit = {
				OR = {
					province_id = 492 #Bilbao
					province_id = 493 #Pamplona
				}
			}
			any_pop = {
				militancy = 1
				consciousness = 1
			}
		}
		ai_chance = {
			factor = 70
		}
	}

	option = {
		name = "EVTOPTB1002700"
		set_country_flag = carlism_questioned
		set_country_flag = carlos_recognised_1830
		prestige = -2
		add_country_modifier = {
			name = national_instability
			duration = 1825
		}
		add_country_modifier = {
			name = conservative_reaction
			duration = 1095
		}
		any_pop = {
			ideology = {
				value = reactionary
				factor = 0.05
			}
			scaled_militancy = {
				ideology = liberal
				factor = 4
			}
		}
		ai_chance = {
			factor = 20
		}
	}

	option = {
		name = "EVTOPTC1002700"
		set_country_flag = carlism_questioned
		set_country_flag = succession_deferred
		treasury = -3000
		add_country_modifier = {
			name = national_instability
			duration = 1825
		}
		any_pop = {
			consciousness = -1
		}
		ai_chance = {
			factor = 10
		}
	}
}
