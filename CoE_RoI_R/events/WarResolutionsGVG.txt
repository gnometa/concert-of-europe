# The Jihad Has Run Its Course
country_event = {
	id = 1002200
	title = "EVTNAME1002200"
	desc = "EVTDESC1002200"
	picture = "war_ended"

	trigger = {
		tag = MAS
		exists = yes
		year = 1843
		NOT = { year = 1848 }
		war_with = SEG
		NOT = {
			has_country_flag = massina_peace_offered
			has_country_flag = massina_settled
		}
	}

	mean_time_to_happen = {
		months = 6
		modifier = {
			factor = 0.5
			war_exhaustion = 10
		}
	}

	option = {
		name = "EVTOPTA1002200" # Make peace with the Bamana
		set_country_flag = massina_peace_offered
		set_country_flag = massina_settled
		prestige = 5
		war_exhaustion = -10
		any_country = {
			limit = {
				tag = SEG
				war_with = MAS
			}
			end_war = MAS
			relation = { who = MAS value = 50 }
		}
		any_pop = {
			limit = { is_primary_culture = yes }
			militancy = -1
		}
		ai_chance = {
			factor = 70
		}
	}

	option = {
		name = "EVTOPTB1002200" # Seku Amadu will not sheathe the sword
		set_country_flag = massina_peace_offered
		set_country_flag = massina_fight_on
		prestige = 2
		war_exhaustion = 5
		any_pop = {
			limit = { is_primary_culture = yes }
			militancy = 1
		}
		ai_chance = {
			factor = 30
		}
	}
}
