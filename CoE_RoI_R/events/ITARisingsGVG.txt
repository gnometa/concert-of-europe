# The Carbonari
country_event = {
	id = 1001100
	title = "EVTNAME1001100"
	desc = "EVTDESC1001100"
	picture = "national_congress"

	fire_only_once = yes

	trigger = {
		OR = {
			tag = SAR
			tag = SIC
			tag = PAP
			tag = TUS
			tag = MOD
			tag = PAR
			tag = LUC
		}
		exists = yes
		year = 1821
		NOT = { year = 1836 }
		NOT = { has_country_flag = carbonari_active }
	}

	mean_time_to_happen = {
		months = 30
		modifier = {
			factor = 0.6
			FRA = { has_country_flag = july_revolution }
		}
		modifier = {
			factor = 0.75
			militancy = 2
		}
	}

	option = {
		name = "EVTOPTA1001100"
		set_country_flag = carbonari_active
		any_pop = {
			consciousness = 1
		}
		capital_scope = {
			add_province_modifier = {
				name = liberal_agitation
				duration = 1095
			}
		}
		ai_chance = {
			factor = 40
		}
	}

	option = {
		name = "EVTOPTB1001100"
		set_country_flag = carbonari_active
		prestige = -1
		add_country_modifier = {
			name = conservative_reaction
			duration = 1825
		}
		any_pop = {
			limit = {
				OR = {
					type = officers
					type = bureaucrats
					type = artisans
				}
			}
			militancy = 1
		}
		ai_chance = {
			factor = 60
		}
	}
}

# Ciro Menotti's Conspiracy
country_event = {
	id = 1001101
	title = "EVTNAME1001101"
	desc = "EVTDESC1001101"
	picture = "streetriot"

	fire_only_once = yes

	trigger = {
		tag = MOD
		exists = yes
		owns = 739
		year = 1830
		NOT = { year = 1836 }
		NOT = { has_country_flag = mod_menotti_resolved }
		OR = {
			year = 1831
			FRA = { has_country_flag = july_revolution }
		}
	}

	mean_time_to_happen = {
		months = 6
		modifier = {
			factor = 0.5
			has_country_flag = carbonari_active
		}
		modifier = {
			factor = 0.75
			FRA = { has_country_flag = july_revolution }
		}
	}

	option = {
		name = "EVTOPTA1001101"
		set_country_flag = mod_menotti_resolved
		prestige = 1
		relation = { who = AUS value = 25 }
		add_country_modifier = {
			name = conservative_reaction
			duration = 1825
		}
		any_pop = {
			militancy = 2
			consciousness = 1
		}
		ai_chance = {
			factor = 70
		}
	}

	option = {
		name = "EVTOPTB1001101"
		set_country_flag = mod_menotti_resolved
		set_global_flag = italian_risings_1831
		prestige = -2
		relation = { who = AUS value = -25 }
		any_owned = {
			limit = {
				OR = {
					province_id = 739 #Modena
					province_id = 740 #Massa
				}
			}
			add_province_modifier = {
				name = liberal_agitation
				duration = 730
			}
		}
		any_pop = {
			consciousness = 2
			militancy = 1
		}
		ai_chance = {
			factor = 30
		}
	}
}
