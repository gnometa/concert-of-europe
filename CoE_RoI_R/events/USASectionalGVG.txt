# The Monroe Doctrine
country_event = {

	id = 1001200

	title = "EVTNAME1001200"
	desc = "EVTDESC1001200"
	picture = "greatpowers"

	major = yes

	trigger = {
		tag = USA
		year = 1823
		NOT = { year = 1828 }
	}

	fire_only_once = yes

	mean_time_to_happen = {
		months = 3
	}

	option = {
		name = "EVTOPTA1001200" # America for the Americans
		prestige = 10
		any_country = {
			limit = {
				exists = yes
				NOT = { tag = USA }
				NOT = { has_country_flag = monroe_doctrine2 }
				OR = {
					continent = north_america
					continent = south_america
				}
			}
			relation = {
				who = USA
				value = 15
			}
			set_country_flag = monroe_doctrine2
		}
		ai_chance = {
			factor = 85
		}
	}

	option = {
		name = "EVTOPTB1001200" # The Old World is not our concern
		prestige = -5
		any_pop = {
			consciousness = -1
		}
		ai_chance = {
			factor = 15
		}
	}
}
