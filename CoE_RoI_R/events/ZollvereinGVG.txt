# Accession to the Zollverein (the German minor)
country_event = {
	id = 1000900
	title = "EVTNAME1000900"
	desc = "EVTDESC1000900"
	picture = "national_congress"
	is_triggered_only = yes
	major = yes
	news = yes
	news_title = "EVTNAME1000900_NEWS_TITLE"
	news_desc_long = "EVTDESC1000900_NEWS_LONG"
	news_desc_medium = "EVTDESC1000900_NEWS_MEDIUM"
	news_desc_short = "EVTDESC1000900_NEWS_SHORT"

	option = {
		name = "EVTOPTA1000900"
		set_country_flag = zollverein_member
		# read by 1000904 and by decisions/ZollvereinGVG.txt: any_country cannot be
		# used as a trigger, so the "somebody joined / somebody refused" tests are
		# global flags set here instead.
		set_global_flag = zollverein_has_member
		add_country_modifier = {
			name = zollverein_member
			duration = -1
		}
		prestige = 2
		relation = { who = FROM value = 50 }
		relation = { who = AUS value = -25 }
		FROM = {
			diplomatic_influence = { who = THIS value = 100 }
			country_event = { id = 1000901 days = 30 }
		}
		any_pop = {
			limit = { is_primary_culture = yes }
			consciousness = 1
		}
		ai_chance = {
			factor = 55
			modifier = {
				factor = 3
				in_sphere = FROM
			}
			modifier = {
				factor = 2
				relation = { who = FROM value = 100 }
			}
			modifier = {
				factor = 1.5
				NOT = { relation = { who = AUS value = 50 } }
			}
			modifier = {
				factor = 0.25
				in_sphere = AUS
			}
			modifier = {
				factor = 0.5
				has_country_flag = south_german_rel
			}
		}
	}

	option = {
		name = "EVTOPTB1000900"
		set_country_flag = zollverein_refused
		# read by form_the_southern_customs_union: only a SOUTH German refusal gives
		# Vienna something to build on. Vic2 has no if/else, so the conditional set is
		# the random_owned + limit idiom.
		random_owned = {
			limit = { owner = { has_country_flag = south_german_rel } }
			owner = { set_global_flag = zollverein_refusal }
		}
		prestige = 5
		relation = { who = FROM value = -25 }
		relation = { who = AUS value = 25 }
		AUS = { diplomatic_influence = { who = THIS value = 50 } }
		FROM = { country_event = { id = 1000902 days = 60 } }
		ai_chance = {
			factor = 45
			modifier = {
				factor = 3
				in_sphere = AUS
			}
			modifier = {
				factor = 0.2
				in_sphere = FROM
			}
			modifier = {
				factor = 0.5
				relation = { who = FROM value = 100 }
			}
		}
	}
}

# The German Customs Union is complete (the founder)
country_event = {
	id = 1000904
	title = "EVTNAME1000904"
	desc = "EVTDESC1000904"
	picture = "national_congress"
	major = yes
	news = yes
	news_title = "EVTNAME1000904_NEWS_TITLE"
	news_desc_long = "EVTDESC1000904_NEWS_LONG"
	news_desc_medium = "EVTDESC1000904_NEWS_MEDIUM"
	news_desc_short = "EVTDESC1000904_NEWS_SHORT"

	fire_only_once = yes

	trigger = {
		has_global_flag = zollverein_founded
		has_country_flag = zollverein_founder
		NOT = { has_global_flag = zollverein_complete }
		# "at least one state joined". any_country is an EFFECT scope: used as a
		# trigger the engine eats a brace and this file loses the events after it,
		# so the accession is recorded as a global flag in 1000900 option A instead.
		has_global_flag = zollverein_has_member
	}

	# The old trigger also required that no invitee was still undecided, which
	# needs a country iterator no trigger scope provides. The offers are answered
	# within 30-60 days (1000900 -> 1000901/1000902), so the wait is a timer now.
	mean_time_to_happen = {
		months = 6
	}

	option = {
		name = "EVTOPTA1000904"
		set_global_flag = zollverein_complete
		prestige = 10
		any_country = {
			limit = { has_country_flag = zollverein_member }
			relation = { who = THIS value = 10 }
			any_pop = {
				limit = { is_primary_culture = yes }
				consciousness = 1
			}
		}
		any_pop = {
			limit = { is_primary_culture = yes }
			consciousness = 1
			militancy = -1
		}
	}
}
