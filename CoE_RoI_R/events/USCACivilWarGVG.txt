# 1003600: The Federal Rupture
country_event = {
	id = 1003600
	title = "EVTNAME1003600"
	desc = "EVTDESC1003600"
	picture = "churchmexico"

	trigger = {
		tag = UCA
		exists = yes
		year = 1826
		NOT = { year = 1829 }
		NOT = {
			has_country_flag = uca_first_civil_war_begun
			has_country_flag = join_mexico
		}
	}

	mean_time_to_happen = {
		months = 3
	}

	option = {
		name = "EVTOPTA1003600"
		set_country_flag = uca_first_civil_war_begun
		set_country_flag = arce_centralist_coup
		prestige = -2
		any_pop = {
			limit = {
				OR = {
					location = { province_id = 2191 } # San Salvador
					location = { province_id = 2193 } # Comayagua
				}
			}
			militancy = 3
			consciousness = 3
		}
		ai_chance = {
			factor = 90
		}
	}

	option = {
		name = "EVTOPTB1003600"
		set_country_flag = uca_first_civil_war_begun
		prestige = 2
		any_pop = {
			limit = {
				location = { province_id = 2186 } # Guatemala
			}
			militancy = 2
		}
		ai_chance = {
			factor = 10
		}
	}
}

# 1003605: The Disintegration of the Federal Republic
country_event = {
	id = 1003605
	title = "EVTNAME1003605"
	desc = "EVTDESC1003605"
	picture = "deliberation"

	trigger = {
		tag = UCA
		exists = yes
		has_country_flag = carrera_rebellion_active
		year = 1838
		month = 4 # May
		NOT = { year = 1842 }
		NOT = { has_country_flag = uca_dissolution_completed }
	}

	mean_time_to_happen = {
		days = 30
	}

	option = {
		name = "EVTOPTA1003605"
		set_country_flag = uca_dissolution_completed
		prestige = -15
		any_owned = {
			limit = { is_core = NIC }
			secede_province = NIC
		}
		any_owned = {
			limit = { is_core = HON }
			secede_province = HON
		}
		any_owned = {
			limit = { is_core = COS }
			secede_province = COS
		}
		any_owned = {
			limit = { is_core = GUA }
			secede_province = GUA
		}
		any_owned = {
			limit = { is_core = ELS }
			secede_province = ELS
		}
		# 2181 Tuxtla has MEX/CHP/UCA cores only - without this the federation
		# leaves a one-province rump UCA that has lost its capital. Chiapas proper
		# voted to join Mexico in 1824; Soconusco (2182) has a GUA core and goes to GUA.
		any_owned = {
			limit = { province_id = 2181 }
			secede_province = MEX
		}
		ai_chance = {
			factor = 85
		}
	}

	option = {
		name = "EVTOPTB1003605"
		set_country_flag = uca_dissolution_completed
		prestige = -5
		any_pop = {
			militancy = 4
		}
		ai_chance = {
			factor = 15
		}
	}
}
