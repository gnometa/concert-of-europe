		tag = SWE

		year = 1821

# 1003402: The Battle of the Square (Torvslaget, 17 May 1829)
country_event = {
	id = 1003402
	title = "EVTNAME1003402"
	desc = "EVTDESC1003402"
	picture = "streetriot"

	trigger = {
		tag = NOR
		exists = yes
		vassal_of = SWE
		year = 1829
		month = 4
		NOT = { year = 1832 }
		NOT = { has_country_flag = torvslaget_happened }
		# The riot happened because Christiania celebrated in defiance of the ban;
		# a Storting that complied in 1003401 option B gets no Torvslaget.
		NOT = { has_country_flag = complied_17_may }
	}

	mean_time_to_happen = {
		days = 17
	}

	option = {
		name = "EVTOPTA1003402"
		set_country_flag = torvslaget_happened
		set_global_flag = torvslaget_scandal
		any_pop = {
			militancy = 2
			consciousness = 3
		}
		313 = {
			add_province_modifier = {
				name = liberal_agitation
				duration = 1095
			}
		}
		SWE = {
			prestige = -5
			relation = { who = NOR value = -40 }
		}
		ai_chance = {
			factor = 100
		}
	}
}
