	trigger = {
		tag = NET
		exists = yes
		owns = 1416
		owns = 1413
		year = 1825
		NOT = { year = 1831 }
		NOT = { has_country_flag = java_war_in_progress }
		NOT = { has_country_flag = java_war_settled }
		NOT = { has_country_flag = cultuurstelsel }
	}

	mean_time_to_happen = {
		months = 3
	}

	option = {
		name = "EVTOPTA1001300"
		set_country_flag = java_war_in_progress
		prestige = -3
		treasury = -20000
		war_exhaustion = 3
		any_owned = {
			limit = {
				OR = {
					province_id = 1413
					province_id = 1414
					province_id = 1415
					province_id = 1416
					province_id = 1417
					province_id = 1418
					province_id = 1419
					province_id = 1420
					province_id = 1421
				}
			}
			add_province_modifier = {
				name = java_war
				duration = 1825
			}
			any_pop = {
				militancy = 2
				consciousness = 1
			}
		}
		country_event = { id = 1001301 days = 1825 }
		ai_chance = {
			factor = 80
		}
	}

	option = {
		name = "EVTOPTB1001300"
		set_country_flag = java_war_settled
		prestige = -5
		treasury = -5000
		any_owned = {
			limit = {
				OR = {
					province_id = 1416
					province_id = 1419
				}
			}
			add_province_modifier = {
				name = java_war
				duration = 730
			}
		}
		country_event = { id = 1001302 days = 60 }
		ai_chance = {
			factor = 15
			modifier = {
				factor = 2
				war = yes
			}
		}
	}

	option = {
		name = "EVTOPTC1001300"
		set_country_flag = java_war_in_progress
		set_country_flag = java_war_neglected
		prestige = -8
		treasury = -5000
		any_owned = {
			limit = {
				OR = {
					province_id = 1413

					province_id = 1417

				militancy = 3
				consciousness = 1
			}
