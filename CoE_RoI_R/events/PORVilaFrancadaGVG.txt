#########################################################
#                                                       #
#   Vila-Francada and the Abrilada, 1823-1826 (GVG)     #
#   ids 1003000-1003004                                 #
#                                                       #
#   The collapse of the Vintista regime. Hands off to   #
#   events/PORFlavor.txt 97030 (Death of John VI,       #
#   1825) and from there to PORMiguelistGVG.txt         #
#   1001000. This chain never sets por_charter_granted, #
#   miguelist_usurpation, por_cortes_resists or         #
#   por_charter_upheld - those belong to 97030 and to   #
#   the Miguelist file.                                 #
#                                                       #
#   1003004 is the 1826 Carta: it puts Portugal back    #
#   under hms_government on the branches that ended     #
#   absolutist or prussian, so the later PORFlavor      #
#   chains (97000/97001/97003/97020) stay reachable.    #
#                                                       #
#   UPB is the start-date tag; 97024 turns it into POR, #
#   so every trigger reads OR = { tag = POR tag = UPB }.#
#                                                       #
#########################################################

# The Revolt in Tras-os-Montes
country_event = {
	id = 1003000
	title = "EVTNAME1003000"
	desc = "EVTDESC1003000"
	picture = "Rebellion"

	fire_only_once = yes

		OR = { tag = POR tag = UPB }
		exists = yes
		year = 1823
		NOT = { year = 1824 }
		NOT = { has_country_flag = por_vintismo_challenged }

	mean_time_to_happen = {
		months = 3

			factor = 0.5
			FRA = { has_country_flag = one_hunderd_thousand_sons }

		modifier = {
			factor = 0.75
			SPA = { government = absolute_monarchy }
		}
		modifier = {
			factor = 0.75
			ruling_party_ideology = reactionary
		}
		modifier = {
			factor = 0.75
			average_militancy = 3
		}
	}

	# March on Vila Real
	option = {
		name = "EVTOPTA1003000"
		set_country_flag = por_vintismo_challenged
		treasury = -5000
		prestige = 1
		add_country_modifier = {
			name = military_strife
			duration = 365
		}
		any_owned = {
			limit = {
				OR = {
					province_id = 519 #Vila Real
					province_id = 520 #Covilha
				}
			}
			any_pop = {
				militancy = 2
				consciousness = 1
				ideology = {
					value = reactionary
					factor = 0.10
				}
			}
		}
		country_event = { id = 1003001 days = 90 }
		ai_chance = {
			factor = 55
		}
	}

	# Amnesty for the Count of Amarante
	option = {
		name = "EVTOPTB1003000"
		set_country_flag = por_vintismo_challenged

			}

				value = reactionary

		}
		any_owned = {
			limit = {

# The Carta Constitucional
# Only fires where the Vila-Francada left Portugal ruling without the Cortes
# (1003001 option A or B, 1003002 option B). The branch that kept the
# constitution (1003001 option C) is already hms_government and never sees it.
# Both options end in hms_government: without that, every later PORFlavor
# event gated on hms_government (97000, 97001, 97003, 97020) is dead content.
country_event = {
	id = 1003004
	title = "EVTNAME1003004"
	desc = "EVTDESC1003004"
	picture = "Monarchy"

	trigger = {
		OR = { tag = POR tag = UPB }
		exists = yes
		year = 1825
		NOT = { year = 1836 }
		has_country_flag = por_charter_granted
		NOT = { has_country_flag = por_carta_restored }
		NOT = { has_country_flag = miguelist_usurpation }
		OR = {
			government = absolute_monarchy
			government = absolute_monarchy2
			government = absolute_monarchy3
			government = prussian_constitutionalism
			government = prussian_constitutionalism2
			government = prussian_constitutionalism3
		}
	}

	mean_time_to_happen = {
		months = 2
		modifier = {
			factor = 0.75
			has_country_flag = por_british_guarantee
		}
		modifier = {
			factor = 0.75
			has_country_flag = por_miguel_exiled
		}
	}

	option = {
		name = "EVTOPTA1003004"

			limit = { has_province_modifier = liberal_agitation }
			remove_province_modifier = liberal_agitation
		}

			ideology = {
				value = conservative
				factor = 0.10

				ideology = liberal
