########################################################################
# ECONOMY PULSE HOST
#
# Several of this mod's book-keeping systems are written as ordinary
# country events whose effects are global (any_country = { ... }).  They
# must therefore be evaluated for exactly ONE country, or every effect
# would be applied once per country in the world.  The original code did
# that by gating them on tag = BHU, which silently killed the whole
# economy rework the moment Bhutan was annexed (very likely by 1860).
#
# The pattern used instead: one arbitrary surviving AI country carries the
# country flag "economy_pulse_host".  Consumers gate on
#     trigger = { has_country_flag = economy_pulse_host }
# which behaves exactly like the old tag gate (one country, evaluated
# daily, so MTTH rates are unchanged) but survives annexation.
#
# 2000100 elects the host, and is the ONLY moving part: it is self-healing,
# so there is no watchdog and nothing has to be added to on_actions.
# Its trigger is
#     is_greater_power = yes
#     ai = yes
#     NOT = { any_greater_power = { has_country_flag = economy_pulse_host } }
# The greater-power gate keeps the daily cost to at most 8 evaluations of
# the any_country scan (~200 countries), and the scan is false - so it
# short-circuits after the first hit - on every day a host exists, which
# is every day but the first and the day after a host is annexed.
# `ai = yes` keeps the (invisible book-keeping) host off the human player.
# The option strips economy_pulse_host from any previous holder before
# setting its own, so even a same-day double election ends with exactly
# one host.  Latency: the host dies -> a new one is elected the next day;
# the work it does is then picked up at the next pulse it is due on.
#
# An earlier version used a global flag economy_pulse_lock plus a
# watchdog event 2000101 listed on on_quarterly_pulse.  That was wrong:
# on_actions fires exactly ONE entry from the list per country per pulse,
# so listing the watchdog alongside 99997 halved the rate of the entire
# quarterly economy pulse.  on_quarterly_pulse contains only 99997 again.
#
# Users of the flag: 99997 (below), +education_RGO.txt 999958,
# WorkPlaceEvents_triggers.txt 12000-12780.
########################################################################

country_event = {
	id = 2000100
	title = "Economy Pulse Host"
	desc = "Economy Pulse Host"

	trigger = {
		is_greater_power = yes
		ai = yes
		# any_country is an EFFECT scope; as a trigger it eats a brace and drops the
		# next event from the file. any_greater_power is the trigger-side equivalent
		# and the host is a greater power by the line above.
		NOT = { any_greater_power = { has_country_flag = economy_pulse_host } }
	}

	mean_time_to_happen = {
		days = 1
	}

	option = {
		name = "OK"
		# Idempotent: strip the flag from any previous host first, so a re-election can
		# never leave two countries carrying it (which would run every global effect twice).
		any_country = {
			limit = { has_country_flag = economy_pulse_host }
			clr_country_flag = economy_pulse_host
		}
		set_country_flag = economy_pulse_host
	}
}

country_event = {
	id = 99997
	title = "POPs Maintenance"
	desc = "POPs Maintenance"

    is_triggered_only = yes

	trigger = {
		has_country_flag = economy_pulse_host
	}

	mean_time_to_happen = {
		days = 1
	}

	option = {
		name = "Keep the books."
		any_country = { limit = { any_owned_province = { has_province_modifier = settlement_colony } } set_country_flag = land_suitable_for_settlement }
		any_country = { limit = { NOT = { any_owned_province = { has_province_modifier = settlement_colony } } } clr_country_flag = land_suitable_for_settlement }
		any_country = { limit = { war_with = BHU } end_war = BHU }
		# Release the 99988 "Enemy Lies Fallen" once-per-war guard once the war is over.
		any_country = { limit = { has_country_flag = enemy_lies_fallen war = no } clr_country_flag = enemy_lies_fallen }
		#any_country = { add_tax_relative_income = 1.0 }
		# Anti-hoarding drain -- DISABLED, see always = no below.
		#
		# Intent: the Vic2 AI never spends down a large surplus, so by the late game
		# several majors sit on huge treasuries, outbid everyone on the world market and
		# make the great-power score meaningless.  This quarterly sink burned the excess
		# back out of the game.
		#
		# Why it is off: every money value in Paradox script is parsed as fixed point,
		# int32 hundredths, so the largest number that survives the parser is 2^31/100 =
		# 2147483 (docs/wiki/list-of-effects.md, "treasury").  The nine- and ten-figure
		# constants below are far past that in BOTH halves of each line:
		#   - the treasury effect wraps, so -50000000 can PAY the country instead of
		#     draining it;
		#   - the money trigger wraps the same way, so "money = 100000000" actually tests
		#     something like "money >= 14100654" -- it never meant what it reads as.
		# Nothing anywhere in vanilla, PDM or this mod uses a money value above 1000000,
		# so there is no precedent to copy for a working large-treasury test.
		#
		# Rewriting the effect as repeated "treasury = -2000000" fixes only half of it;
		# the brackets themselves cannot be expressed above 2147483, and redesigning them
		# around a 2M threshold would drain most of the world rather than a few hoarders.
		# Picking real thresholds needs the economy balance pass that also owns
		# production_types.txt and goods.txt, so the sink is parked until then.
		any_country = {
			limit = {
				always = no
				money = 150000000  NOT = { money = 200000000 }
			}
			treasury = -50000000
		}
		any_country = {
			limit = {
				always = no
				money = 200000000  NOT = { money = 300000000 }
			}
			treasury = -100000000
		}
		any_country = {
			limit = {
				always = no
				money = 300000000  NOT = { money = 500000000 }
			}
			treasury = -200000000
		}
		any_country = {
			limit = {
				always = no
				money = 500000000  NOT = { money = 1000000000 }
			}
			treasury = -400000000
		}
		any_country = {
			limit = {
				always = no
				money = 1000000000
			}
			treasury = -900000000
		}
		BHU = { any_pop = { money = -100000 } }
		#any_pop = { limit = { country = { NOT = { education_spending = 0.01 } } literacy = -0.01 } }
		#any_country = { limit = { NOT = { education_spending = 0.01 } } any_pop = { literacy = -0.01 } }
		random_country = {
			limit = { 
				2127 = { owner = { NOT = { has_country_flag = ghost_country } } }
				exists = no 
				OR = { 
					is_mobilised = yes 
					#total_amount_of_ships = 1 
				} 
			}
			country_event = 6016821
		}		
		
		# Pop-type retirement.  This mod does not use aristocrats, 00_urban_poor or
		# clergymen: they are folded into farmers, craftsmen and clerks (and colonial
		# non-accepted bureaucrats into capitalists).  The poptypes still exist as files
		# because the promotion tables in poptypes/*.txt still list them as promotion
		# TARGETS (artisans/craftsmen/labourers -> aristocrats, artisans/craftsmen ->
		# clergymen, clerks/farmers -> 00_urban_poor are all live), so pops keep being
		# created in them every month.  The rewrite therefore has to be CONTINUOUS -- a
		# one-off migration at game start would be undone within a year.  Do not gate it
		# on a flag unless those promotion entries are commented out first.
		#
		# The identical block used to appear twice, once here in the host country's own
		# scope and once inside any_country below; the host copy was redundant (the
		# host is one of the any_country members) and has been removed.
		any_country = {
			any_pop = {
				limit = { 
					type = aristocrats 
				}
				pop_type = farmers
			}
			any_pop = { 
				limit = { 
					type = 00_urban_poor 
				}
				pop_type = craftsmen
			}
			any_pop = { 
				limit = { 
					type = clergymen 
				}
				pop_type = clerks
			}	
			any_pop = { 
				limit = { 
					type = bureaucrats 
					location = { is_colonial = yes }
					is_primary_culture = no
					is_accepted_culture = no
				}
				pop_type = capitalists
			}			
			#any_pop = { 
			#	limit = { 
			#		type = clerks 
			#		is_primary_culture = no
