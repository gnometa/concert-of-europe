# The Great Emigration
country_event = {
	id = 1002605
	title = "EVTNAME1002605"
	desc = "EVTDESC1002605"
	picture = "emigration"

	trigger = {
		tag = RUS
		has_country_flag = cpl_uprising
		NOT = { war_with = CPL }
		# The rising must actually have been put down - same gate as PDM 95073.
		OR = {
			NOT = { exists = CPL }
			CPL = { vassal_of = THIS }
			AND = {
				exists = CPL
				truce_with = CPL
			}
		}
		NOT = { has_country_flag = great_emigration }
		year = 1831
		NOT = { year = 1836 }
	}

	mean_time_to_happen = {
		months = 3
		modifier = {
			factor = 0.75
			has_country_flag = november_capitulation
		}
	}

	# Confiscation, courts-martial and the road to Siberia
	option = {
		name = "EVTOPTA1002605"
		set_country_flag = great_emigration
		prestige = 2
		treasury = 5000
		any_pop = {
			limit = {
				culture = polish
			}
			consciousness = 2
			militancy = 1
		}
		any_country = {
			limit = {
				OR = {
					tag = FRA
					tag = ENG
				}
				exists = yes
			}
			relation = { who = RUS value = -25 }
		}
		# nicholas_reaction is owned by PDM 95073 (The Organic Statute), which
		# fires in the same window for 3650 days. Applying it here as well only
		# races 95073 and shortens the reaction, so it is deliberately absent.
		ai_chance = {
			factor = 75
			modifier = {
				factor = 2
				has_country_flag = november_capitulation
			}
		}
	}

	# Let them go, and amnesty those who stayed
	option = {
		name = "EVTOPTB1002605"
		set_country_flag = great_emigration
		prestige = -2
		treasury = -2000
		any_pop = {
			limit = {
				culture = polish
			}
			militancy = -4
			consciousness = -1
		}
		relation = { who = FRA value = 10 }
		relation = { who = ENG value = 10 }
		ai_chance = {
			factor = 25
		}
	}
}
