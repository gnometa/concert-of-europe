	id = 1003500
	title = "EVTNAME1003500"
	desc = "EVTDESC1003500"
	picture = "oldsoldiers"

	trigger = {
		OR = {
			tag = SAR
			tag = SIC
		}
		exists = yes
		year = 1821
		NOT = { year = 1825 }
		NOT = { has_country_flag = austrian_occupation_aftermath }
	}

	mean_time_to_happen = {
		months = 2
	}

	option = {
		name = "EVTOPTA1003500"
		set_country_flag = austrian_occupation_aftermath
		prestige = 2
		relation = { who = AUS value = 40 }

			limit = {
				OR = {
					type = officers

			}

		}

			factor = 85
