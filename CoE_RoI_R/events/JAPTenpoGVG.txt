# The Tenpo Famine - 1833-1837
country_event = {

	id = 1002002

	title = "EVTNAME1002002"
	desc = "EVTDESC1002002"
	picture = "starving"

	trigger = {
		tag = TKG
		exists = yes
		year = 1833
		NOT = { year = 1838 }
	}

	fire_only_once = yes

	mean_time_to_happen = {
		months = 6
	}

	option = {
		name = "EVTOPTA1002002" # Open the granaries of Edo
		money = -15000
		any_pop = {
			militancy = 1
		}
		random_owned = {
			add_province_modifier = {
				name = famine
				duration = 730
			}
			any_pop = { reduce_pop = 0.97 }
		}
		random_owned = {
			add_province_modifier = {
				name = famine
				duration = 730
			}
			any_pop = { reduce_pop = 0.97 }
		}
		set_country_flag = tkg_tenpo_relief
		ai_chance = {
			factor = 40
			modifier = {
				factor = 0
				NOT = { money = 15000 }
			}
		}
	}

	option = {
		name = "EVTOPTB1002002" # The domains must feed themselves
		any_owned = {
			add_province_modifier = {
				name = famine
				duration = 730
			}
			any_pop = {
				militancy = 2
				reduce_pop = 0.95
			}
		}
		set_country_flag = tkg_tenpo_neglect
		random = {
			chance = 50
			country_event = {
				id = 1002003
				days = 1250
			}
		}
		ai_chance = {
			factor = 60
		}
	}
}
