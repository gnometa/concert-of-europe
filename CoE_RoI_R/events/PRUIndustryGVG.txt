# 1003700: Krupp's Cast Steel Works in Essen
country_event = {
	id = 1003700
	title = "EVTNAME1003700"
	desc = "EVTDESC1003700"
	picture = "factory"

	trigger = {
		tag = PRU
		exists = yes
		owns = 578 # Dusseldorf / Ruhr
		year = 1826
		NOT = { year = 1830 }
		NOT = { has_country_flag = krupp_gussstahlfabrik }
	}

	mean_time_to_happen = {
		months = 6
	}

	option = {
		name = "EVTOPTA1003700"
		set_country_flag = krupp_gussstahlfabrik
		prestige = 2
		578 = {
			add_province_modifier = {
				name = local_iron_works
				duration = 3650
			}
		}
		any_pop = {
			limit = {
				location = { province_id = 578 }
				OR = {
					type = craftsmen
					type = capitalists
				}
			}
			consciousness = 1
		}
		ai_chance = {
			factor = 95
		}
	}

	option = {
		name = "EVTOPTB1003700"
		set_country_flag = krupp_gussstahlfabrik
		prestige = -1
		ai_chance = {
			factor = 5
		}
	}
}
